cmake_minimum_required(VERSION 3.20)
project(vn_engine CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL3 REQUIRED CONFIG)
add_library(vn_core STATIC src/vnm.cpp src/renderer.cpp)
target_include_directories(vn_core PUBLIC src third_party)
target_link_libraries(vn_core PUBLIC SDL3::SDL3)
if(APPLE)
  target_link_libraries(vn_core PUBLIC "-framework OpenGL")
  target_compile_definitions(vn_core PUBLIC GL_SILENCE_DEPRECATION)
endif()
add_executable(vn_viewer src/viewer.cpp)
target_link_libraries(vn_viewer PRIVATE vn_core)
